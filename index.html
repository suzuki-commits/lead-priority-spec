実装コード

"""生成AI活用支援・顧問への問い合わせ優先度を決める。"""

import re


def _to_int(value, default):
    # "300名" "約500名" "1,200" "50〜99"（範囲は下限）などから最初の整数を取り出す
    if value is None or isinstance(value, bool):
        return default
    match = re.search(r"-?\d+", str(value).replace(",", ""))
    return int(match.group()) if match else default


def _to_bool(value):
    if isinstance(value, str):
        return value.strip().lower() in {"true", "1", "yes", "はい", "希望"}
    return bool(value)


def lead_priority(lead):
    score, reasons = 0, []
    employees = _to_int(lead.get("employees"), 0)
    if employees >= 300:
        score += 3
        reasons.append("従業員300名以上")
    elif employees >= 50:
        score += 2
        reasons.append("従業員50名以上")
    if _to_bool(lead.get("requested_demo")):
        score += 3
        reasons.append("デモ希望")
    if _to_bool(lead.get("budget_confirmed")):
        score += 2
        reasons.append("予算確認済み")
    if (_to_bool(lead.get("first_meeting_done"))
            and not _to_bool(lead.get("second_meeting_scheduled"))):
        score += 3
        reasons.append("初回面談済み・2回目未設定")
    days = _to_int(lead.get("days_since_contact"), 999)
    if 0 <= days <= 7:
        score += 1
        reasons.append("直近7日以内の接点")
    priority = "HIGH" if score >= 6 else "MEDIUM" if score >= 3 else "LOW"
    return {"priority": priority, "score": score, "reasons": reasons}
